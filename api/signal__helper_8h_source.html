18 template <typename T>
19 struct UniversalCallback {
20 static void callback(GObject* obj, gpointer user_data) {
21 // Interpret user_data as vector
22 auto data_vector = static_cast<std::vector<void*>*>(user_data);
23
24 // Extract our pair
25 auto pair = static_cast<std::pair<T*, void (T::*)(GObject*, void*)>*>(data_vector->at(0));
26
27 // optional extra data
28 void* extra_data = (data_vector->size() > 1) ? data_vector->at(1) : nullptr;
29
30 // Call method
31 (pair->first->*(pair->second))(obj, extra_data);
32 }
33
34 static void destroy(gpointer user_data, GClosure* closure) {
35 auto data_vector = static_cast<std::vector<void*>*>(user_data);
36
37 // Clean up pair
38 delete static_cast<std::pair<T*, void (T::*)(GObject*, void*)>*>(data_vector->at(0));
39
40 // Clean up wrapper data
41 delete data_vector;
42 }
